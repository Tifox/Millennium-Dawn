# Estonian Leader Rotation Effects for Elections
# These effects create the appropriate Prime Minister based on the game date when an election occurs

# Create Estonian Reform Party Leader (Liberalism)
# Reform Party PMs: Siim Kallas (2002-2003), Andrus Ansip (2005-2014), Taavi Roivas (2014-2016), Kaja Kallas (2021+)
EST_create_reform_leader = {
	if = {
		limit = { date < 2005.4.1 }
		# Siim Kallas (2002-2003)
		create_country_leader = {
			name = "Siim Kallas"
			desc = EST_siim_kallas_desc
			picture = "siim_kallas.dds"
			expire = "2050.1.1"
			ideology = liberalism
			traits = {
				western_liberalism
				economist
				pro_american
			}
		}
	}
	else_if = {
		limit = {
			date > 2005.4.1
			date < 2014.3.1
		}
		# Andrus Ansip (2005-2014)
		create_country_leader = {
			name = "Andrus Ansip"
			desc = EST_andrus_ansip_desc
			picture = "Andrus_Ansip.dds"
			expire = "2050.1.1"
			ideology = liberalism
			traits = {
				western_liberalism
				career_politician
				pro_american
				tech_savy
			}
		}
	}
	else_if = {
		limit = {
			date > 2014.3.1
			date < 2021.1.1
		}
		# Taavi Roivas (2014-2016) - also used for Juri Ratas coalition period
		create_country_leader = {
			name = "Taavi Roivas"
			desc = EST_taavi_roivas_desc
			picture = "Taavi_Roivas.dds"
			expire = "2050.1.1"
			ideology = liberalism
			traits = {
				western_liberalism
				inexperienced
				pro_american
			}
		}
	}
	else = {
		# Kaja Kallas (2021+)
		create_country_leader = {
			name = "Kaja Kallas"
			desc = EST_kaja_kallas_desc
			picture = "kaja_kallas.dds"
			expire = "2050.1.1"
			ideology = liberalism
			traits = {
				western_liberalism
				lawyer
				pro_american
			}
		}
	}
}

EST_random_ai = {
	hidden_effect = {
		if = {
			limit = {
				EST = {
					is_ai = yes
				}
			}
			if = {
				limit = {
					has_global_flag = EST_FASCIST
				}
				if = {
					limit = { NOT = { is_in_array = { ruling_party = 21 } } }
					set_temp_variable = { rul_party_temp = 21 }
					change_ruling_party_effect = yes
				}
				set_temp_variable = { party_index = 21 }
				set_temp_variable = { party_popularity_increase = 0.35 }
				set_temp_variable = { temp_outlook_increase = 0.35 }
				add_relative_party_popularity = yes
			}
			if = {
				limit = {
					has_global_flag = EST_CENTER
				}
				if = {
					limit = { NOT = { is_in_array = { ruling_party = 15 } } }
					set_temp_variable = { rul_party_temp = 15 }
					change_ruling_party_effect = yes
				}
				set_temp_variable = { party_index = 15 }
				set_temp_variable = { party_popularity_increase = 0.35 }
				set_temp_variable = { temp_outlook_increase = 0.35 }
				add_relative_party_popularity = yes
			}
			if = {
				limit = {
					has_global_flag = EST_DEMOCRATIC
				}
				random_list = {
					50 = {
						if = {
							limit = { NOT = { is_in_array = { ruling_party = 2 } } }
							set_temp_variable = { rul_party_temp = 2 }
							change_ruling_party_effect = yes
							set_politics = {
								ruling_party = democratic
								elections_allowed = yes
							}
						}
						set_temp_variable = { party_index = 2 }
						set_temp_variable = { party_popularity_increase = 0.35 }
						set_temp_variable = { temp_outlook_increase = 0.35 }
						add_relative_party_popularity = yes
					}
					50 = {
						if = {
							limit = { NOT = { is_in_array = { ruling_party = 3 } } }
							set_temp_variable = { rul_party_temp = 3 }
							change_ruling_party_effect = yes
							set_politics = {
								ruling_party = democratic
								elections_allowed = yes
							}
						}
						set_temp_variable = { party_index = 3 }
						set_temp_variable = { party_popularity_increase = 0.35 }
						set_temp_variable = { temp_outlook_increase = 0.35 }
						add_relative_party_popularity = yes
					}
				}
			}
		}
	}
}
